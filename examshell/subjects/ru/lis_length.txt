Напишите функцию, которая возвращает длину самой длинной СТРОГО
ВОЗРАСТАЮЩЕЙ подпоследовательности массива int: элементы не обязаны
идти подряд, но должны сохранять исходный относительный порядок. Для
пустого массива возвращается 0.

Примеры:
    lis_length([10,9,2,5,3,7,101,18], 8) -> 4   ([2,3,7,101] or [2,3,7,18])
    lis_length([7,7,7,7], 4)             -> 1

    int lis_length(int *arr, unsigned int size);
