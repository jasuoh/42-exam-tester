Напишите функцию, которая находит элементы, присутствующие в КАЖДОМ
из данных списков целых чисел, и возвращает их без повторов,
отсортированными по возрастанию.

    def list_intersection_finder(lists: list[list[int]]) -> list[int]:

Верните пустой список, если внешний список пуст, если любой из
внутренних списков пуст или если общих элементов нет вовсе.

Примеры:
    list_intersection_finder([[1,2,3], [2,3,4], [2,3,5]]) -> [2, 3]
    list_intersection_finder([[1,2,3,4], [2,4,6,8], [4,8,12]]) -> [4]
    list_intersection_finder([[1,1,2,3], [1,2,2,3], [1,2,3,3]]) -> [1, 2, 3]
    list_intersection_finder([[1,2,3], [4,5,6]])          -> []
    list_intersection_finder([])                          -> []
    list_intersection_finder([[1,2,3], []])               -> []
    list_intersection_finder([[5]])                       -> [5]
